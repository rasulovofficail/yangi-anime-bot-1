async def init_db():

async def add_user(user_id: int, username: str | None, full_name: str | None):

# ==================== ANIME API (JIKAN.MOE) ====================
async def search_anime(query: str, limit: int = 4):
    url = f"https://api.jikan.moe/v4/anime?q={query}&limit={limit}"
    try:
        async with aiohttp.ClientSession() as session:
            async with session.get(url, timeout=aiohttp.ClientTimeout(total=10)) as resp:
                if resp.status == 200:
                    data = await resp.json()
                    return data.get("data", [])
                elif resp.status == 429:
                    logger.warning("Jikan API: Rate limit (429)")
    except Exception as e:
        logger.error(f"Search API xatosi: {e}")
    return []

def format_anime(anime: dict) -> str:
    """Xavfsiz HTML formatlash - Telegram syntax crashlarini oldini oladi"""
    title = html.escape(anime.get("title") or "Noma'lum")
    title_jp = html.escape(anime.get("title_japanese") or "")
    score = html.escape(str(anime.get("score") or "—"))
    episodes = html.escape(str(anime.get("episodes") or "?"))
    status = html.escape(str(anime.get("status") or "—"))
    year = html.escape(str(anime.get("year") or "—"))
    
    genre_list = [g.get("name", "") for g in anime.get("genres", []) if g.get("name")]
    genres = html.escape(", ".join(genre_list[:4]) or "—")
    
    raw_synopsis = anime.get("synopsis") or "Tavsif mavjud emas"
    safe_synopsis = html.escape(raw_synopsis[:280].strip()) + ("..." if len(raw_synopsis) > 280 else "")

    text = f"""<b>🎌 {title}</b>
{f'<i>{title_jp}</i>' if title_jp else ''}

⭐ <b>Reyting:</b> {score}
📺 <b>Epizodlar soni:</b> {episodes}
📅 <b>Yil:</b> {year}
🟢 <b>Status:</b> {status}
🎭 <b>Janr:</b> {genres}

📝 <b>Tavsif:</b>
{safe_synopsis}"""
    return text.strip()

@dp.message(BotStates.waiting_search)
async def process_search(message: Message, state: FSMContext):

    if query.startswith("/"):
        if query == "/cancel":
            await state.clear()
            await message.answer("❌ Qidiruv bekor qilindi.", reply_markup=main_menu())
            return

    status_msg = await message.answer("⏳ Qidirilmoqda...")

    results = await search_anime(query)
    try:
        await status_msg.delete()
    except Exception:
        pass

    if not results:
        await message.answer(
            f"❌ '<b>{html.escape(query)}</b>' bo‘yicha hech qanday anime topilmadi.\n"
            "Iltimos, nomini to'g'ri yozganingizni tekshiring.",
            reply_markup=main_menu()
        )
        await state.clear()
        return

    for anime in results[:3]:
        text = format_anime(anime)
        image = anime.get("images", {}).get("jpg", {}).get("large_image_url")
        url = anime.get("url") or "https://myanimelist.net"
        keyboard = InlineKeyboardMarkup(inline_keyboard=[
            [InlineKeyboardButton(text="🌐 MyAnimeList da ko'rish", url=url)]
        ])
        if image:
            try:
                await message.answer_photo(photo=image, caption=text[:1000], reply_markup=keyboard)
            except Exception:
                await message.answer(text, reply_markup=keyboard)
        else:
            await message.answer(text, reply_markup=keyboard)

    await message.answer("🔍 Yana qidirish yoki menyuga qaytish uchun:", reply_markup=main_menu())
    await state.clear()

        try:
            await bot.copy_message(chat_id=user_id, from_chat_id=message.chat.id, message_id=message.message_id)

            await asyncio.sleep(0.05)  # Telegram flood limitiga qarshi

            fail += 1

    await status_msg.edit_text(f"✅ <b>Natija:</b>\n\nYuborildi: {success} ta\nYetib bormadi: {fail} ta")
    await state.clear()
